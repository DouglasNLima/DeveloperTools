    function normaliseGuid(value) {
        if (!value) return null;

        const guid = String(value)
            .replace(/[{}]/g, "")
            .trim()
            .toLowerCase();

        return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/
            .test(guid)
            ? guid
            : null;
    }

    function escapeODataString(value) {
        return String(value).replace(/'/g, "''");
    }

    function toApiUrl(pathOrUrl) {
        if (!pathOrUrl) {
            throw new Error("Empty Dataverse request URL.");
        }

        const value = String(pathOrUrl);

        if (/^https?:\/\//i.test(value)) {
            const absolute = new URL(value);

            if (absolute.origin !== location.origin) {
                throw new Error(
                    `Refusing cross-origin forensic GET: ${absolute.href}`
                );
            }

            return absolute.href;
        }

        // Runtime WebResource URLs are application-root routes, not Web API routes.
        // Preserve them exactly so /webresources/... never becomes
        // /api/data/v9.2/webresources/....
        if (value.startsWith("/webresources/")) {
            return value;
        }

        if (value.startsWith("webresources/")) {
            return `/${value}`;
        }

        if (value.startsWith("/api/data/")) {
            return value;
        }

        if (value.startsWith("api/data/")) {
            return `/${value}`;
        }

        const path = value.startsWith("/")
            ? value
            : `/${value}`;

        return `${apiBase}${path}`;
    }

    function xhrGet(pathOrUrl, options = {}) {
        const {
            allow404 = false,
            asArrayBuffer = false,
            timeoutMs = CONFIG.requestTimeoutMs
        } = options;

        const url = toApiUrl(pathOrUrl);

        return new Promise((resolve, reject) => {
            const xhr = new XMLHttpRequest();

            xhr.open("GET", url, true);
            xhr.timeout = timeoutMs;

            const resolvedUrl = new URL(url, location.origin);
            const isDataverseApi =
                resolvedUrl.pathname.startsWith("/api/data/");

            if (isDataverseApi) {
                xhr.setRequestHeader("OData-MaxVersion", "4.0");
                xhr.setRequestHeader("OData-Version", "4.0");
            }

            xhr.setRequestHeader(
                "Accept",
                asArrayBuffer ? "*/*" : "application/json"
            );

            if (asArrayBuffer) {
                xhr.responseType = "arraybuffer";
            }

            xhr.onload = () => {
                let body = null;

                if (asArrayBuffer) {
                    body = xhr.response ?? new ArrayBuffer(0);
                } else {
                    const text = xhr.responseText ?? "";

                    if (text) {
                        try {
                            body = JSON.parse(text);
                        } catch {
                            body = text;
                        }
                    }
                }

                const ok = xhr.status >= 200 && xhr.status < 300;

                if (!ok && !(allow404 && xhr.status === 404)) {
                    reject(
                        new Error(
                            `${xhr.status} ${xhr.statusText}\n` +
                            `${url}\n` +
                            (
                                typeof body === "string"
                                    ? body
                                    : JSON.stringify(body)
                            )
                        )
                    );
                    return;
                }

                resolve({
                    ok,
                    status: xhr.status,
                    statusText: xhr.statusText,
                    body,
                    headers: {
                        contentType: xhr.getResponseHeader("content-type")
                    },
                    url
                });
            };

            xhr.onerror = () => {
                reject(
                    new Error(
                        `Network error while requesting:\n${url}`
                    )
                );
            };

            xhr.ontimeout = () => {
                reject(
                    new Error(
                        `Request timed out after ${timeoutMs} ms:\n${url}`
                    )
                );
            };

            xhr.onabort = () => {
                reject(
                    new Error(
                        `Request aborted:\n${url}`
                    )
                );
            };

            xhr.send();
        });
    }

    async function request(pathOrUrl, options = {}) {
        const retries =
            options.retries ?? CONFIG.requestRetries;

        let lastError = null;

        for (
            let attempt = 0;
            attempt <= retries;
            attempt++
        ) {
            const url = toApiUrl(pathOrUrl);

            console.debug(
                `[PCF FORENSICS] GET ${attempt + 1}/${retries + 1}: ${url}`
            );

            try {
                return await xhrGet(pathOrUrl, options);
            } catch (error) {
                lastError = error;

                if (attempt < retries) {
                    console.warn(
                        `[PCF FORENSICS] GET failed; retrying: ${url}`,
                        error
                    );
                }
            }
        }

        throw lastError ?? new Error(
            `GET failed: ${toApiUrl(pathOrUrl)}`
        );
    }

    async function requiredQuery(label, pathOrUrl) {
        try {
            const result = await request(pathOrUrl);

            return Array.isArray(result.body?.value)
                ? result.body.value
                : [];
        } catch (error) {
            report.errors.push({
                section: label,
                path: pathOrUrl,
                error: String(error)
            });

            throw new Error(
                `Required Dataverse query failed (${label}): ${String(error)}`
            );
        }
    }

    async function query(label, pathOrUrl) {
        try {
            const result = await request(pathOrUrl);

            return Array.isArray(result.body?.value)
                ? result.body.value
                : [];
        } catch (error) {
            report.errors.push({
                section: label,
                path: pathOrUrl,
                error: String(error)
            });

            console.warn(
                `[PCF FORENSICS] Non-blocking query failure: ${label}`,
                error
            );

            return [];
        }
    }

    async function getAll(
        label,
        pathOrUrl,
        maxRows = 1000
    ) {
        const rows = [];
        let next = pathOrUrl;

        try {
            while (
                next &&
                rows.length < maxRows
            ) {
                const page = await request(next);

                if (Array.isArray(page.body?.value)) {
                    rows.push(...page.body.value);
                }

                next =
                    page.body?.["@odata.nextLink"] ??
                    null;
            }
        } catch (error) {
            report.errors.push({
                section: label,
                path: pathOrUrl,
                error: String(error)
            });

            console.warn(
                `[PCF FORENSICS] Non-blocking paged query failure: ${label}`,
                error
            );
        }

        return rows.slice(0, maxRows);
    }

    function hasValue(value) {
        if (value === null || value === undefined) {
            return false;
        }

        if (typeof value === "string") {
            return value.length > 0;
        }

        return true;
    }

    function deriveIdentity(controlName) {
        const underscore = controlName.indexOf("_");

        const schemaIdentity =
            underscore >= 0
                ? controlName.slice(underscore + 1)
                : controlName;

        const lastDot =
            schemaIdentity.lastIndexOf(".");

        if (
            lastDot <= 0 ||
            lastDot === schemaIdentity.length - 1
        ) {
            throw new Error(
                `Unable to derive PCF namespace/constructor from '${controlName}'. ` +
                `Expected a Dataverse CustomControl name similar to ` +
                `'publisher_namespace.Control'.`
            );
        }

        const namespace =
            schemaIdentity.slice(0, lastDot);

        const constructor =
            schemaIdentity.slice(lastDot + 1);

        const canonicalPrefix =
            `cc_${namespace}.${constructor}/`;

        return {
            controlName,
            schemaIdentity,
            namespace,
            constructor,
            identityToken: constructor,
            canonicalPrefix,
            expectedBundleName:
                `${canonicalPrefix}bundle.js`
        };
    }

    }

    function deriveExpectedRuntimeName(
        resourceName,
        identity
    ) {
        if (!resourceName) return null;

        const trimmed =
            String(resourceName).replace(/^\/+/, "");

        if (
            trimmed.startsWith(
                identity.canonicalPrefix
            )
        ) {
            return trimmed;
        }

        if (/bundle\.js$/i.test(trimmed)) {
            return identity.expectedBundleName;
        }

        if (trimmed.includes("/")) {
            return (
                `${identity.canonicalPrefix}${trimmed}`
            );
        }

        return null;
    }

    function addFinding(

    function uniqueBy(items, keySelector) {
        return Array.from(
            new Map(
                items.map(
                    item => [
                        keySelector(item),
                        item
                    ]
                )
            ).values()
        );
    }

    const objectMap = new Map();

    function addObject(
        label,
        id,
        expectedComponentType = null
    ) {
        const guid = normaliseGuid(id);
        if (!guid) return;

        let item =
            objectMap.get(guid);

        if (!item) {
            item = {
                id: guid,
                labels: [],
                expectedComponentTypes: []
            };

            objectMap.set(guid, item);
        }

        if (!item.labels.includes(label)) {
            item.labels.push(label);
        }

        if (
            expectedComponentType !== null &&
            !item.expectedComponentTypes.includes(
                expectedComponentType
            )
        ) {
            item.expectedComponentTypes.push(
                expectedComponentType
            );
        }
    }

    // =========================================================

    function buildSummary() {
        const control = report.control;

        const criticalFindings =
            report.findings.filter(
                x => x.severity === "CRITICAL"
            );

        const warningFindings =
            report.findings.filter(
                x => x.severity === "WARNING"
            );

        const danglingCount =
            report.referencedWebResources.filter(
                x => x.exists === false
            ).length;

        const payloadDefectCount =
            report.referencedWebResources.filter(
                x =>
                    x.exists === true &&
                    !(
                        x.contentPresent ||
                        x.contentfilerefPresent
                    )
            ).length;

        const httpHealthyCount =
            report.httpResults.filter(
                x =>
                    x.ok === true &&
                    x.bytes > 0
            ).length;

        const resourceVersions = [
            ...new Set(
                report.resources
                    .map(x => x.version)
                    .filter(Boolean)
            )
        ];

        report.summary = {
            controlId:
                control?.customcontrolid ?? null,

            controlName:
                control?.name ??
                CONFIG.controlName,

            controlVersion:
                control?.version ?? null,

            controlManaged:
                control?.ismanaged ?? null,

            controlState:
                control?.componentstate ?? null,

            controlCreatedOn:
                control?.createdon ?? null,

            controlModifiedOn:
                control?.modifiedon ?? null,

            customControlResourceCount:
                report.resources.length,

            resourceVersions,

            referencedWebResourcesResolved:
                report.referencedWebResources
                    .filter(x => x.exists === true)
                    .length,

            danglingWebResourceLinks:
                danglingCount,

            referencedPayloadDefects:
                payloadDefectCount,

            canonicalWebResourceCount:
                report.canonicalWebResources.length,

            broadWebResourceMatchCount:
                report.broadWebResourceMatches.length,

            expectedRuntimeResourceCount:
                report.expectedWebResources.length,

            httpTests:
                report.httpResults.length,

            httpHealthy:
                httpHealthyCount,

            solutionComponentRows:
                report.solutionComponents.length,

            componentLayerRowsById:
                report.componentLayersById.length,

            componentLayerRowsByName:
                report.componentLayersByName.length,

            solutionsResolved:
                report.solutions.length,

            oldestSolutionHistoryAvailableUtc:
                report.solutionHistoryRetention
                    .oldestAvailableUtc,

            newestSolutionHistoryAvailableUtc:
                report.solutionHistoryRetention
                    .newestAvailableUtc,

            relatedSolutionHistoryRows:
                report.relatedSolutionHistory.length,

            criticalFindings:
                criticalFindings.length,

            warningFindings:
                warningFindings.length,

            nonBlockingErrors:
                report.errors.length,

            classification:
                criticalFindings.length > 0
                    ? "INTEGRITY_DEFECT_DETECTED"
                    : warningFindings.length > 0
                        ? "NO_CRITICAL_DEFECT_WITH_WARNINGS"
                        : "HEALTHY"
        };
    }

    function finalise(
        terminalMessage = null,
        terminalState = null
    ) {
        if (!report.summary) {
            buildSummary();
        }

        window.__PCF_FORENSICS = report;

        console.log("=== PCF FORENSIC FINDINGS ===");
        console.table(report.findings);

        console.log("=== PCF FORENSIC SUMMARY ===");
        console.table([report.summary]);

        if (report.errors.length > 0) {
            console.log("=== NON-BLOCKING QUERY ERRORS ===");
            console.table(report.errors);
        }

        if (CONFIG.printFinalJson) {
            console.log("=== PCF FORENSICS JSON START ===");
            console.log(
                JSON.stringify(report, null, 2)
            );
            console.log("=== PCF FORENSICS JSON END ===");
        }

        console.log(
            "Full object available as window.__PCF_FORENSICS"
        );

        enableUiActions();

        const state =
            terminalState ??
            (
                report.summary.classification === "HEALTHY"
                    ? "success"
                    : report.summary.classification ===
                        "NO_CRITICAL_DEFECT_WITH_WARNINGS"
                        ? "warning"
                        : "error"
            );

        setUiStatus(
            terminalMessage ??
            `Completed: ${report.summary.classification}`,
            state
        );

        if (ui?.note) {

            uniqueBy(
                relatedHistory,
                x =>
                    x.msdyn_solutionhistoryid ??
                    [
                        x.msdyn_starttime,
                        x.msdyn_name,
                        x.msdyn_correlationid
                    ].join("|")
            );

        console.log(
            "=== 10B. RELATED RETAINED SOLUTION HISTORY ==="
        );

        console.table(
            report.relatedSolutionHistory
        );

        // -----------------------------------------------------
        // 11. MISSING GUID TRACE
        // -----------------------------------------------------
        setProgress("11/12 Tracing missing WebResource GUIDs…");

        const missingReferencedResources =
            report.referencedWebResources.filter(
                x =>
                    x.exists === false &&
                    normaliseGuid(
                        x.webresourceid
                    )
            );

        for (
            const missing
            of missingReferencedResources
        ) {
            const id =
                normaliseGuid(
                    missing.webresourceid
                );

            const resource =
                report.resources.find(
                    x =>
                        x.customcontrolresourceid ===
                        missing.customcontrolresourceid
                );

            const expected =
                report.expectedWebResources.find(
                    x =>
                        x.customcontrolresourceid ===
                        missing.customcontrolresourceid
                );

            const solutionComponentRows =
                report.solutionComponents.filter(
                    x =>
                        normaliseGuid(
                            x.objectid
                        ) === id
                );

            const layersById =
                report.componentLayersById.filter(
                    x =>
                        normaliseGuid(
                            x.queriedObjectId
                        ) === id ||
                        normaliseGuid(
                            x.msdyn_componentid
                        ) === id
                );

            const layersByName =
                report.componentLayersByName.filter(
                    x =>
                        normaliseGuid(
                            x.msdyn_componentid
                        ) === id
                );

            report.missingGuidTrace.push({
                customcontrolresourceid:
                    missing.customcontrolresourceid,

                resourceName:
                    missing.resourceName,

                resourceVersion:
                    resource?.version ??
                    null,

                missingWebResourceId:
                    id,

                directWebResourceStatus:
                    missing.status,

                solutionComponentRows:
                    solutionComponentRows.length,

                componentLayersById:
                    layersById.length,

                componentLayersByName:
                    layersByName.length,

                expectedRuntimeName:
                    expected?.expectedRuntimeName ??
                    null,

                expectedRuntimeResourceExists:
                    expected
                        ?.expectedRuntimeResourceExists ??
                    false,

                expectedRuntimeWebResourceId:
                    expected
                        ?.expectedRuntimeWebResourceId ??
                    null,

                recreatedWithDifferentId:
                    expected
                        ?.expectedUsesDifferentId ??
                    false
            });
        }

        console.log(
            "=== 11. MISSING WEBRESOURCE GUID TRACE ==="
        );

        console.table(
            report.missingGuidTrace
        );

        // -----------------------------------------------------
        // 12. FINDINGS
        // -----------------------------------------------------
        setProgress("12/12 Classifying findings…");

        if (
            report.resources.length === 0
        ) {
            addFinding(
                "NO_CUSTOM_CONTROL_RESOURCES",
                "CRITICAL",
                "The CustomControl has no CustomControlResource rows."
            );
        }

        const resourceVersions = [
            ...new Set(
                report.resources
                    .map(
                        x => x.version
                    )
                    .filter(Boolean)
            )
        ];

        if (
            resourceVersions.length > 0 &&
            (
                !resourceVersions.includes(
                    control.version
                ) ||
                resourceVersions.length > 1
            )
        ) {
            addFinding(
                "RESOURCE_VERSION_DRIFT",
                "WARNING",
                "CustomControl and CustomControlResource versions are not fully aligned.",
                {
                    controlVersion:
                        control.version,

                    resourceVersions
                }
            );
        }

        for (
            const resource
            of report.resources
        ) {
            const direct =
                report.referencedWebResources.find(
                    x =>
                        x.customcontrolresourceid ===
                        resource.customcontrolresourceid
                );

            const expected =
                report.expectedWebResources.find(
                    x =>
                        x.customcontrolresourceid ===
                        resource.customcontrolresourceid
                );

            if (
                !normaliseGuid(
                    resource.webresourceid
                )
            ) {
                addFinding(
                    "MISSING_WEBRESOURCE_ID",
                    "CRITICAL",
                    `CustomControlResource '${resource.name}' has no valid WebResourceId.`,
                    {
                        customcontrolresourceid:
                            resource.customcontrolresourceid
                    }
                );

                continue;
            }

            if (
                direct?.exists === false
            ) {
                addFinding(
                    "DANGLING_WEBRESOURCE_REFERENCE",
                    "CRITICAL",
                    `CustomControlResource '${resource.name}' references a WebResource that does not exist.`,
                    {
                        customcontrolresourceid:
                            resource.customcontrolresourceid,

                        webresourceid:
                            resource.webresourceid,

                        status:
                            direct.status
                    }
                );
            }

            if (
                direct?.exists === true &&
                !(
                    direct.contentPresent ||
                    direct.contentfilerefPresent
                )
            ) {
                addFinding(
                    "WEBRESOURCE_PAYLOAD_DEFECT",
                    "CRITICAL",
                    `Referenced WebResource '${direct.name}' has neither content nor contentfileref.`,
                    {
                        webresourceid:
                            direct.webresourceid,

                        name:
                            direct.name
                    }
                );
            }

            if (
                expected?.expectedRuntimeName &&
                !expected
                    .expectedRuntimeResourceExists
            ) {
                addFinding(
                    "CANONICAL_RUNTIME_RESOURCE_MISSING",
                    "CRITICAL",
                    `Expected runtime WebResource '${expected.expectedRuntimeName}' was not found.`,
                    {
                        customcontrolresourceid:
                            resource.customcontrolresourceid,

                        resourceName:
                            resource.name
                    }
                );
            }

            if (
                expected
                    ?.expectedUsesDifferentId
            ) {
                addFinding(
                    "STALE_WEBRESOURCE_LINK",
                    "CRITICAL",
                    `CustomControlResource '${resource.name}' points to a different WebResourceId than the runtime resource currently found by canonical name.`,
                    {
                        referencedWebResourceId:
                            resource.webresourceid,

                        canonicalWebResourceId:
                            expected
                                .expectedRuntimeWebResourceId,

                        expectedRuntimeName:
                            expected
                                .expectedRuntimeName
                    }
                );
            }

            if (
                direct?.exists === true &&
                resource.ismanaged !==
                    direct.ismanaged
            ) {
                addFinding(
                    "MANAGEMENT_STATE_MISMATCH",
                    "WARNING",
                    `CustomControlResource '${resource.name}' and its referenced WebResource have different ismanaged values.`,
                    {
                        customControlResourceManaged:
                            resource.ismanaged,

                        webResourceManaged:
                            direct.ismanaged,

                        webresourceid:
                            direct.webresourceid
                    }
                );
            }
        }

        if (CONFIG.httpTest) {
            for (
                const expected
                of report.expectedWebResources
            ) {
                if (
                    !expected
                        .expectedRuntimeName
                ) {
                    continue;
                }

                const http =
                    report.httpResults.find(
                        x =>
                            x.name ===
                            expected
                                .expectedRuntimeName
                    );

                if (
                    !http ||
                    !http.ok ||
                    http.bytes <= 0
                ) {
                    addFinding(
                        "HTTP_SERVING_DEFECT",
                        "CRITICAL",
                        `Runtime WebResource '${expected.expectedRuntimeName}' did not return a healthy HTTP response.`,
                        {
                            status:
                                http?.status ??
                                null,

                            bytes:
                                http?.bytes ??
                                0
                        }
                    );
                }
            }
        }

        const oldestHistoryUtc =
            report.solutionHistoryRetention
                .oldestAvailableUtc;

        if (
            oldestHistoryUtc &&
            control.createdon &&
            new Date(
                control.createdon
            ) <
            new Date(
                oldestHistoryUtc
            )
        ) {
            addFinding(
                "CREATION_EVENT_OUTSIDE_HISTORY_RETENTION",
                "INFO",
                "The CustomControl predates the oldest currently retained msdyn_solutionhistories record.",
                {
                    controlCreatedOn:
                        control.createdon,

                    oldestHistoryAvailableUtc:
                        oldestHistoryUtc
                }
            );
        }

        const oldestRelevantTimestamp = [
            control.createdon,
            ...report.resources.map(
                x => x.createdon
            )
        ]
            .filter(Boolean)
            .sort()[0] ??
            null;

        if (
            oldestHistoryUtc &&
            oldestRelevantTimestamp &&
            new Date(
                oldestRelevantTimestamp
            ) <
            new Date(
                oldestHistoryUtc
            )
        ) {
            addFinding(
                "EARLY_COMPONENT_EVENTS_OUTSIDE_HISTORY_RETENTION",
                "INFO",
                "One or more inspected PCF component creation events predate the oldest retained solution history."
            );
        }

        if (
            report.componentLayersById
                .length === 0
        ) {
            addFinding(
                "NO_COMPONENT_LAYERS_RETURNED_BY_ID",
                "INFO",
                "msdyn_componentlayers returned no rows for the inspected object GUIDs. This is not treated as proof that no internal layers exist."
            );
        }

        // Build final report.
        buildSummary();

        return finalise();
    } catch (error) {
        report.errors.push({
            section:
                "FATAL",

            error:
                String(error),

            stack:
                error?.stack ??
                null
        });

        addFinding(
            "FORENSIC_SCRIPT_FATAL_ERROR",
            "CRITICAL",
            "The forensic script encountered a fatal execution error.",
            {
                error:
                    String(error)
            }
        );

        buildSummary();

        console.error(
            "[PCF FORENSICS] Fatal error",
            error
        );

        return finalise(
            `Stopped by error: ${String(error)}`,
            "error"
        );
    }
})();
