    document.documentElement.appendChild(host);
    const q=id=>host.querySelector('#'+id); q('close').onclick=()=>host.remove();
    let diff=null;
    const read=f=>new Promise((resolve,reject)=>{const r=new FileReader();r.onload=()=>{try{resolve(JSON.parse(r.result))}catch(e){reject(e)}};r.onerror=reject;r.readAsText(f)});
    q('go').onclick=async()=>{try{const fa=q('a').files[0],fb=q('b').files[0];if(!fa||!fb)throw new Error('Select both JSON files.');const A=await read(fa),B=await read(fb);diff=(() => {
    const roles=x=>x?.evidence?.roles||[];
    function flatten(input){const out=new Map();for(const r of input){const key=`${r.name}|${r.businessUnitId||''}`;out.set(key,{role:r,priv:new Map((r.privileges||[]).map(p=>[p.name||p.privilegeid,p]))});}return out;}
    const am=flatten(roles(A)),bm=flatten(roles(B));const onlyA=[...am.keys()].filter(k=>!bm.has(k)),onlyB=[...bm.keys()].filter(k=>!am.has(k)),changed=[];
    for(const k of am.keys()){if(!bm.has(k))continue;const ap=am.get(k).priv,bp=bm.get(k).priv;const removed=[...ap.keys()].filter(x=>!bp.has(x)),added=[...bp.keys()].filter(x=>!ap.has(x)),depthChanges=[];for(const n of ap.keys()){if(bp.has(n)&&ap.get(n).depthMask!==bp.get(n).depthMask)depthChanges.push({privilege:n,a:ap.get(n).depthMask,b:bp.get(n).depthMask});}if(removed.length||added.length||depthChanges.length)changed.push({role:k,removedPrivileges:removed,addedPrivileges:added,depthChanges});}
    return {tool:'Security-Role-Diff',capturedUtc:new Date().toISOString(),environmentA:A.environment||null,environmentB:B.environment||null,onlyA,onlyB,changed,summary:{rolesOnlyA:onlyA.length,rolesOnlyB:onlyB.length,rolesChanged:changed.length}};
})();window.__DYNAMICS_FORENSICS_DIFF=diff;console.log('=== FORENSICS DIFF ===');console.log(diff);q('status').textContent=JSON.stringify(diff.summary,null,2);q('download').disabled=false;}catch(e){q('status').textContent=String(e);console.error(e)}};
    q('download').onclick=()=>{if(!diff)return;const blob=new Blob([JSON.stringify(diff,null,2)],{type:'application/json'});const u=URL.createObjectURL(blob);const a=document.createElement('a');a.href=u;a.download=`${TOOL.replace(/[^a-zA-Z0-9._-]/g,'_')}-${new Date().toISOString().replace(/[:.]/g,'-')}.json`;a.click();setTimeout(()=>URL.revokeObjectURL(u),5000)};
