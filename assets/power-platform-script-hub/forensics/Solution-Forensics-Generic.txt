/*
 * Solution-Forensics-Generic
 * Version: 1.0.0
 * Dynamics / Dataverse Forensics Toolkit 1.0.0
 *
 * Read-only diagnostic script for an authenticated Dataverse browser session.
 * Change only the CONFIG block for normal use.
 *
 * SAFETY:
 *   - GET requests only.
 *   - No PATCH / POST / PUT / DELETE.
 *   - No repair or persistence actions.
 *   - Findings are diagnostic heuristics, not Microsoft Support determinations.
 */

(async () => {
    "use strict";

    const TOOL = {
        name: "Solution-Forensics-Generic",
        version: "1.0.0",
        toolkitVersion: "1.0.0"
    };

    const CONFIG = {
        "solutionUniqueName": "your_solution_unique_name",
        "apiVersion": "v9.2",
        "requestTimeoutMs": 15000,
        "requestRetries": 1,
        "componentTop": 5000,
        "patchTop": 500,
        "patchWarningThreshold": 20,
        "historyTop": 300
};
    const apiBase = `/api/data/${CONFIG.apiVersion || "v9.2"}`;

    const report = {
        schemaVersion: 1,
        tool: TOOL,
        capturedUtc: new Date().toISOString(),
        environment: location.origin,
        configuration: { ...CONFIG },
        evidence: {},
        findings: [],
        summary: null,
        errors: []
    };

    window.__DYNAMICS_FORENSICS = report;

    const UI_ID = "__dynamicsForensicsToolkitHost";
    let ui = null;

    function createUi() {
        try {
            const existing = document.getElementById(UI_ID);
            if (existing) existing.remove();

            const host = document.createElement("div");
            host.id = UI_ID;
            Object.assign(host.style, {
                all: "initial",
                position: "fixed",
                top: "12px",
                right: "12px",
                zIndex: "2147483647",
                pointerEvents: "auto"
            });

            const root = host.attachShadow ? host.attachShadow({ mode: "open" }) : host;
            const wrapper = document.createElement("div");
            wrapper.innerHTML = `
                <style>
                    .box { box-sizing:border-box;width:390px;max-width:calc(100vw - 24px);padding:12px;border:1px solid #666;border-radius:8px;background:#fff;color:#111;box-shadow:0 6px 24px rgba(0,0,0,.28);font-family:"Segoe UI",Arial,sans-serif;font-size:12px;line-height:1.35 }
                    .head { display:flex;align-items:center;justify-content:space-between;gap:8px;margin-bottom:8px }
                    .title { font-weight:700;font-size:13px }
                    .version { color:#666;font-size:11px }
                    .status { margin:6px 0 10px;padding:7px 8px;border-radius:5px;background:#f3f3f3;word-break:break-word }
                    .status[data-state="running"] { background:#fff7dc }
                    .status[data-state="success"] { background:#e8f5e9 }
                    .status[data-state="warning"] { background:#fff3e0 }
                    .status[data-state="error"] { background:#fdecea }
                    .buttons { display:flex;flex-wrap:wrap;gap:6px }
                    button { box-sizing:border-box;min-height:30px;padding:5px 9px;border:1px solid #888;border-radius:5px;background:#f8f8f8;color:#111;font:inherit;cursor:pointer }
                    button:disabled { opacity:.45;cursor:default }
                    .note { margin-top:8px;color:#666;font-size:11px;word-break:break-word }
                </style>
                <div class="box">
                    <div class="head"><div><div class="title">${TOOL.name}</div><div class="version">v${TOOL.version} • Toolkit ${TOOL.toolkitVersion}</div></div><button id="close">×</button></div>
                    <div class="status" id="status" data-state="running">Starting…</div>
                    <div class="buttons"><button id="download" disabled>Download JSON</button><button id="copy" disabled>Copy JSON</button><button id="print" disabled>Print JSON</button></div>
                    <div class="note" id="note">Read-only. Change only CONFIG for normal use.</div>
                </div>`;
            root.appendChild(wrapper);
            (document.documentElement || document.body).appendChild(host);

            const q = id => root.getElementById ? root.getElementById(id) : wrapper.querySelector(`#${id}`);
            const status = q('status'), note = q('note'), download = q('download'), copyButton = q('copy'), printButton = q('print'), close = q('close');
            close.onclick = () => host.remove();
            download.onclick = () => {
                const data = JSON.stringify(window.__DYNAMICS_FORENSICS, null, 2);
                const blob = new Blob([data], { type: 'application/json;charset=utf-8' });
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                const safe = TOOL.name.replace(/[^a-zA-Z0-9._-]/g, '_');
                const stamp = new Date().toISOString().replace(/[:.]/g, '-');
                a.href = url;
                a.download = `${safe}-${location.hostname}-${stamp}.json`;
                a.style.display = 'none';
                (document.body || document.documentElement).appendChild(a);
                a.click(); a.remove(); setTimeout(() => URL.revokeObjectURL(url), 5000);
            };
            copyButton.onclick = async () => {
                const data = JSON.stringify(window.__DYNAMICS_FORENSICS, null, 2);
                try { await navigator.clipboard.writeText(data); copyButton.textContent='Copied'; setTimeout(()=>copyButton.textContent='Copy JSON',1200); }
                catch { console.warn('[DYNAMICS FORENSICS] Clipboard unavailable; use Print JSON.'); }
            };
            printButton.onclick = () => {
                console.log('=== DYNAMICS FORENSICS JSON START ===');
                console.log(JSON.stringify(window.__DYNAMICS_FORENSICS, null, 2));
                console.log('=== DYNAMICS FORENSICS JSON END ===');
            };
            ui = { host, status, note, download, copyButton, printButton };
        } catch (error) {
            console.warn('[DYNAMICS FORENSICS] UI could not be installed.', error);
        }
    }

    function setStatus(message, state='running') {
        console.log(`[DYNAMICS FORENSICS] ${message}`);
        if (ui?.status) { ui.status.textContent=message; ui.status.setAttribute('data-state', state); }
    }

    function enableUi() {
        if (!ui) return;
        ui.download.disabled=false; ui.copyButton.disabled=false; ui.printButton.disabled=false;
        ui.note.textContent = `${location.hostname} • errors=${report.errors.length}`;
    }

    function escapeODataString(value) { return String(value).replace(/'/g, "''"); }
    function normaliseGuid(value) {
        if (!value) return null;
        const v=String(value).replace(/[{}]/g,'').trim().toLowerCase();
        return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/.test(v) ? v : null;
    }
    function hasValue(value) { return value !== null && value !== undefined && (typeof value !== 'string' || value.length > 0); }
    function uniqueBy(items, keySelector) { return Array.from(new Map(items.map(x => [keySelector(x), x])).values()); }

    function toUrl(pathOrUrl) {
        const value=String(pathOrUrl || '');
        if (!value) throw new Error('Empty request URL.');
        if (/^https?:\/\//i.test(value)) {
            const u=new URL(value);
            if (u.origin !== location.origin) throw new Error(`Refusing cross-origin GET: ${u.href}`);
            return u.href;
        }
        if (value.startsWith('/webresources/')) return value;
        if (value.startsWith('/api/data/')) return value;
        const path=value.startsWith('/') ? value : `/${value}`;
        return `${apiBase}${path}`;
    }

    function xhrGet(pathOrUrl, options={}) {
        const { allow404=false, asArrayBuffer=false, timeoutMs=CONFIG.requestTimeoutMs || 15000 } = options;
        const url=toUrl(pathOrUrl);
        return new Promise((resolve,reject) => {
            const xhr=new XMLHttpRequest();
            xhr.open('GET', url, true); xhr.timeout=timeoutMs;
            const resolved=new URL(url, location.origin);
            if (resolved.pathname.startsWith('/api/data/')) { xhr.setRequestHeader('OData-MaxVersion','4.0'); xhr.setRequestHeader('OData-Version','4.0'); }
            xhr.setRequestHeader('Accept', asArrayBuffer ? '*/*' : 'application/json');
            if (asArrayBuffer) xhr.responseType='arraybuffer';
            xhr.onload=() => {
                let body=null;
                if (asArrayBuffer) body=xhr.response ?? new ArrayBuffer(0);
                else {
                    const text=xhr.responseText ?? '';
                    if (text) { try { body=JSON.parse(text); } catch { body=text; } }
                }
                const ok=xhr.status>=200 && xhr.status<300;
                if (!ok && !(allow404 && xhr.status===404)) return reject(new Error(`${xhr.status} ${xhr.statusText}\n${url}\n${typeof body==='string'?body:JSON.stringify(body)}`));
                resolve({ ok, status:xhr.status, statusText:xhr.statusText, body, headers:{ contentType:xhr.getResponseHeader('content-type') }, url });
            };
            xhr.onerror=() => reject(new Error(`Network error: ${url}`));
            xhr.ontimeout=() => reject(new Error(`Timeout after ${timeoutMs} ms: ${url}`));
            xhr.onabort=() => reject(new Error(`Request aborted: ${url}`));
            xhr.send();
        });
    }

    async function request(pathOrUrl, options={}) {
        const retries=options.retries ?? CONFIG.requestRetries ?? 1;
        let last=null;
        for (let i=0;i<=retries;i++) {
            try { return await xhrGet(pathOrUrl, options); }
            catch (error) { last=error; if (i<retries) console.warn('[DYNAMICS FORENSICS] GET failed; retrying.', error); }
        }
        throw last;
    }

    async function query(label, pathOrUrl, required=false) {
        try {
            const r=await request(pathOrUrl);
            return Array.isArray(r.body?.value) ? r.body.value : [];
        } catch (error) {
            report.errors.push({ section:label, path:pathOrUrl, error:String(error) });
            if (required) throw error;
            console.warn(`[DYNAMICS FORENSICS] Non-blocking query failed: ${label}`, error);
            return [];
        }
    }

    async function getEntity(label, entitySet, id, select) {
        const guid=normaliseGuid(id); if (!guid) return null;
        try { const r=await request(`/${entitySet}(${guid})?$select=${select}`, { allow404:true }); return r.ok ? r.body : null; }
        catch (error) { report.errors.push({ section:label, error:String(error) }); return null; }
    }

    function addFinding(code, severity, message, details={}) { report.findings.push({ code, severity, message, ...details }); }

    function classify() {
        const critical=report.findings.filter(x=>x.severity==='CRITICAL').length;
        const warnings=report.findings.filter(x=>x.severity==='WARNING').length;
        return critical ? 'INTEGRITY_DEFECT_DETECTED' : warnings ? 'NO_CRITICAL_DEFECT_WITH_WARNINGS' : 'HEALTHY';
    }

    function finalise(extra={}) {
        const classification=extra.classification || classify();
        report.summary = {
            ...extra,
            criticalFindings: report.findings.filter(x=>x.severity==='CRITICAL').length,
            warningFindings: report.findings.filter(x=>x.severity==='WARNING').length,
            infoFindings: report.findings.filter(x=>x.severity==='INFO').length,
            nonBlockingErrors: report.errors.length,
            classification
        };
        window.__DYNAMICS_FORENSICS=report;
        console.log('=== FINDINGS ==='); console.table(report.findings);
        console.log('=== SUMMARY ==='); console.table([report.summary]);
        enableUi();
        const state=classification==='HEALTHY'||classification==='SNAPSHOT_COMPLETE'?'success':classification.includes('WARNING')||classification.includes('GAPS')?'warning':'error';
        setStatus(`Completed: ${classification}`, state);
        return report;
    }

    createUi();
    try {
        setStatus('1/6 Reading solution…');
        const solutions=await query('solution', `/solutions?$select=solutionid,uniquename,friendlyname,version,ismanaged,installedon,createdon,modifiedon,_parentsolutionid_value&$filter=uniquename eq '${escapeODataString(CONFIG.solutionUniqueName)}'&$top=10`, true);
        report.evidence.solutions=solutions;
        if (solutions.length !== 1) {
            addFinding(solutions.length===0?'SOLUTION_NOT_FOUND':'MULTIPLE_SOLUTIONS_FOUND','CRITICAL',`Expected one solution '${CONFIG.solutionUniqueName}', found ${solutions.length}.`);
            return finalise({ solutionCount:solutions.length });
        }
        const solution=solutions[0]; const solutionId=normaliseGuid(solution.solutionid);
        setStatus('2/6 Reading solution components…');
        const components=await query('solution components', `/solutioncomponents?$select=solutioncomponentid,componenttype,objectid,rootsolutioncomponentid,rootcomponentbehavior,ismetadata,createdon,modifiedon&$filter=_solutionid_value eq ${solutionId}&$top=${CONFIG.componentTop}`);
        report.evidence.solutionComponents=components;
        const typeMap={"1": "Entity", "2": "Attribute", "20": "Security Role", "24": "Form", "26": "Saved Query", "29": "Workflow / Process", "44": "Duplicate Detection Rule", "60": "System Form", "61": "Web Resource", "62": "Site Map", "63": "Connection Role", "66": "Custom Control", "69": "Custom Control Resource", "80": "App Module", "91": "Plug-in Assembly", "92": "SDK Message Processing Step"};
        const counts={}; for (const c of components) { const k=`${c.componenttype} ${typeMap[c.componenttype]||'Unknown'}`; counts[k]=(counts[k]||0)+1; }
        report.evidence.componentTypeCounts=counts;
        setStatus('3/6 Resolving parent and child patches…');
        const parentId=normaliseGuid(solution._parentsolutionid_value);
        report.evidence.parentSolution=parentId ? await getEntity('parent solution','solutions',parentId,'solutionid,uniquename,friendlyname,version,ismanaged,installedon,createdon,modifiedon,_parentsolutionid_value') : null;
        const children=await query('child solutions', `/solutions?$select=solutionid,uniquename,friendlyname,version,ismanaged,installedon,createdon,modifiedon,_parentsolutionid_value&$filter=_parentsolutionid_value eq ${solutionId}&$orderby=installedon asc&$top=${CONFIG.patchTop}`);
        report.evidence.childSolutions=children;
        if (parentId && !report.evidence.parentSolution) addFinding('PARENT_SOLUTION_NOT_FOUND','CRITICAL','The solution references a parent solution that could not be resolved.',{ parentSolutionId:parentId });
        if (children.length > CONFIG.patchWarningThreshold) addFinding('PATCH_CHAIN_COMPLEXITY','WARNING',`The solution has ${children.length} child patches.`,{ childPatchCount:children.length });
        setStatus('4/6 Reading retained solution history…');
        const history=await query('solution history', `/msdyn_solutionhistories?$select=msdyn_solutionhistoryid,msdyn_name,msdyn_solutionversion,msdyn_ismanaged,msdyn_ispatch,msdyn_isoverwritecustomizations,msdyn_operation,msdyn_suboperation,msdyn_status,msdyn_result,msdyn_starttime,msdyn_endtime,msdyn_totaltime,msdyn_activityid,msdyn_correlationid,msdyn_errorcode,msdyn_exceptionmessage&$filter=contains(msdyn_name,'${escapeODataString(CONFIG.solutionUniqueName)}')&$orderby=msdyn_starttime desc&$top=${CONFIG.historyTop}`);
        report.evidence.solutionHistory=history;
        const failed=history.filter(x=>x.msdyn_result===false || (x.msdyn_errorcode && x.msdyn_errorcode!=='0'));
        const overwrite=history.filter(x=>x.msdyn_isoverwritecustomizations===true);
        if (failed.length) addFinding('FAILED_IMPORT_HISTORY','WARNING',`${failed.length} retained solution-history entries indicate a failure or non-zero error code.`,{ count:failed.length });
        if (overwrite.length) addFinding('OVERWRITE_CUSTOMISATIONS_HISTORY','INFO',`${overwrite.length} retained imports used overwrite customisations.`,{ count:overwrite.length });
        setStatus('5/6 Checking component anomalies…');
        const nullObject=components.filter(x=>!x.objectid);
        if (nullObject.length) addFinding('SOLUTION_COMPONENT_WITHOUT_OBJECT_ID','WARNING',`${nullObject.length} SolutionComponent rows have no objectid.`,{ count:nullObject.length });
        setStatus('6/6 Finalising…');
        return finalise({ solutionUniqueName:solution.uniquename, solutionVersion:solution.version, managed:solution.ismanaged, componentCount:components.length, childPatchCount:children.length, retainedHistoryRows:history.length });
    } catch (error) {
        report.errors.push({ section:'FATAL', error:String(error), stack:error?.stack ?? null });
        addFinding('FORENSIC_SCRIPT_FATAL_ERROR','CRITICAL','The script encountered a fatal execution error.',{ error:String(error) });
        console.error('[DYNAMICS FORENSICS] Fatal error', error);
        finalise();
    }
})();
